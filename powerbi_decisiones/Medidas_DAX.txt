// Suma del volumen disponible. Los faltantes no se sustituyen por cero.
Crudo observado (bbl) =
SUM(Produccion[Crudo_bbl])

// Unidad adoptada del diccionario del proyecto; no es crudo equivalente.
Gas observado (miles pies3) =
SUM(Produccion[Gas_miles_pies3])

// Filas unicas activo-mes en el contexto filtrado.
N registros =
COUNTROWS(Produccion)

// Numero de observaciones mensuales no vacias de crudo; por activo equivale al n del grupo.
N crudo =
COUNT(Produccion[Crudo_bbl])

// Numero de observaciones mensuales no vacias de gas.
N gas =
COUNT(Produccion[Gas_miles_pies3])

// Dias de los meses seleccionados dentro de enero 2022-agosto 2026, contados una sola vez. Un activo con cobertura incompleta conserva el denominador del periodo.
Dias del periodo productivo =
CALCULATE(SUM(Calendario[Dias_mes]), KEEPFILTERS(Calendario[Con_produccion] = 1))

// Volumen observado / dias del periodo. No es la media simple de las tasas ni un ajuste de faltantes; revisar n y cobertura al comparar activos.
Aporte diario observado (bbl) =
DIVIDE([Crudo observado (bbl)], [Dias del periodo productivo])

// Media simple entre meses con precio; no se repite por activo, no se pondera por produccion y no equivale a ingresos.
Precio promedio (USD/bbl) =
AVERAGE(Precios[Precio_USD_bbl])

// Numero de precios mensuales validos en el periodo.
N meses con precio =
COUNT(Precios[Precio_USD_bbl])

// Meses de la ventana productiva sin precio: agosto de 2026 en la fuente disponible.
Meses productivos sin precio =
CALCULATE(COUNTROWS(Calendario), KEEPFILTERS(Calendario[Con_produccion] = 1), KEEPFILTERS(Calendario[Con_precio] = 0))

// Activos con al menos un registro dentro de la seleccion.
Activos observados =
DISTINCTCOUNT(Produccion[Activo])

// Celdas de crudo anuladas por valores discrepantes de una misma clave.
Conflictos crudo =
SUM(Produccion[Conflicto_crudo])

// Celdas de gas anuladas por valores discrepantes de una misma clave.
Conflictos gas =
SUM(Produccion[Conflicto_gas])

// Cuenta de celdas conflictivas, no de filas duplicadas.
Conflictos totales =
[Conflictos crudo] + [Conflictos gas]

// Banderas IQR calculadas previamente dentro de cada activo. Se conservan los registros.
Atipicos crudo =
SUM(Produccion[Atipico_crudo])

// Banderas IQR del gas diario dentro de cada activo; el filtro no recalcula los limites historicos.
Atipicos gas =
SUM(Produccion[Atipico_gas])

// Meses ausentes entre el primer y ultimo registro de cada activo; no son volumen cero.
Huecos internos =
CALCULATE(COUNTROWS(Cobertura), KEEPFILTERS(Cobertura[Estado] = "sin registro interior"))

// Umbral descriptivo de doce meses para advertir series cortas; no garantiza independencia ni suficiencia estadistica.
Estado del grupo =
VAR N = [N crudo] RETURN IF(ISBLANK(N) || N = 0, "Sin crudo valido", IF(N < 12, "PRECAUCION: n < 12", "n >= 12; revisar cobertura"))

// Numero de activos seleccionados con entre uno y once meses validos de crudo.
Grupos pequenos =
SUMX(VALUES(Activos[Activo]), VAR N = CALCULATE([N crudo]) RETURN IF(N > 0 && N < 12, 1, 0))

// Advertencia dinamica segun filtros. Observaciones mensuales, no muestras independientes.
Advertencia de grupos =
VAR G = [Grupos pequenos] RETURN IF(ISBLANK([N crudo]) || [N crudo] = 0, "Sin datos de crudo en esta seleccion", IF(G > 0, FORMAT(G, "0") & " grupo(s) con n < 12. Interpretar con cautela.", "Revisar cobertura, faltantes y cambios de composicion."))

// Fraccion del crudo observado entre los activos seleccionados, manteniendo el periodo.
Participacion crudo =
DIVIDE([Crudo observado (bbl)], CALCULATE([Crudo observado (bbl)], ALLSELECTED(Activos)))

// Enero-agosto 2026; mismos activos y 243 días que en 2025. Suma de tasas con denominador idéntico.
Comparable 2026 =
SUM(Comparacion[Bpd_2026])

// Comparable 2025
Comparable 2025 =
SUM(Comparacion[Bpd_2025])

// Cambio comparable bpd
Cambio comparable bpd =
[Comparable 2026] - [Comparable 2025]

// Cambio comparable pct
Cambio comparable pct =
DIVIDE([Cambio comparable bpd], [Comparable 2025])

// Activos comparables
Activos comparables =
COUNTROWS(Comparacion)

// Activos en descenso
Activos en descenso =
COALESCE(CALCULATE(COUNTROWS(Comparacion), KEEPFILTERS(Comparacion[Cambio_bpd] < 0)), 0)

// Activos en aumento
Activos en aumento =
COALESCE(CALCULATE(COUNTROWS(Comparacion), KEEPFILTERS(Comparacion[Cambio_bpd] > 0)), 0)

// Impacto absoluto bpd
Impacto absoluto bpd =
ABS([Cambio comparable bpd])

// N comparable 2026
N comparable 2026 =
SUM(Comparacion[N_2026])

// N comparable 2025
N comparable 2025 =
SUM(Comparacion[N_2025])

// Aumentos bpd
Aumentos bpd =
CALCULATE([Cambio comparable bpd], KEEPFILTERS(Comparacion[Cambio_bpd] >= 0))

// Descensos bpd
Descensos bpd =
CALCULATE([Cambio comparable bpd], KEEPFILTERS(Comparacion[Cambio_bpd] < 0))

// Diario comparable 2025
Diario comparable 2025 =
DIVIDE(SUM(SerieComparable[Crudo_2025]), SUMX(VALUES(SerieComparable[Mes_numero]), CALCULATE(MAX(SerieComparable[Dias_mes]))))

// Diario comparable 2026
Diario comparable 2026 =
DIVIDE(SUM(SerieComparable[Crudo_2026]), SUMX(VALUES(SerieComparable[Mes_numero]), CALCULATE(MAX(SerieComparable[Dias_mes]))))

// Sensibilidad: excluye julio de ambos años; siete meses y 212 días por año. No reemplaza la comparación principal completa.
Variacion sin julio =
VAR A = CALCULATE(SUM(SerieComparable[Crudo_2026]), KEEPFILTERS(SerieComparable[Mes_numero] <> 7)) VAR B = CALCULATE(SUM(SerieComparable[Crudo_2025]), KEEPFILTERS(SerieComparable[Mes_numero] <> 7)) RETURN IF(B>0,DIVIDE(A,B)-1)

// Caida prioritaria
Caida prioritaria =
VAR T = TOPN(1, FILTER(Comparacion, Comparacion[Cambio_bpd] < 0), Comparacion[Cambio_bpd], ASC, Comparacion[Activo], ASC) RETURN IF(COUNTROWS(T)=0,"Sin descensos en selección",CONCATENATEX(T,Comparacion[Activo] & " | " & FORMAT(Comparacion[Cambio_bpd],"#,0") & " bbl/día"))

// Crudo 2026
Crudo 2026 =
CALCULATE([Crudo observado (bbl)], KEEPFILTERS(Calendario[Anio]=2026))

// Diario 2026
Diario 2026 =
CALCULATE([Aporte diario observado (bbl)], KEEPFILTERS(Calendario[Anio]=2026))

// Participacion 2026
Participacion 2026 =
DIVIDE([Crudo 2026], CALCULATE([Crudo 2026], ALLSELECTED(Activos)))

// Participación de SA/AU/SH entre los activos seleccionados. Grupo fijo, no ranking recalculado.
Participacion tres activos =
DIVIDE(CALCULATE([Crudo 2026], KEEPFILTERS(Activos[Grupo_volumen]="SA + AU + SH")), [Crudo 2026])

// Activos 2026
Activos 2026 =
CALCULATE([Activos observados], KEEPFILTERS(Calendario[Anio]=2026))

// N 2026
N 2026 =
CALCULATE([N crudo], KEEPFILTERS(Calendario[Anio]=2026))

// Once activos completos durante 56 meses. No depende del filtro Activos.
Diario cohorte =
DIVIDE(SUM(Cohorte[Crudo_bbl]), SUM(Cohorte[Dias_mes]))

// N pares mensuales
N pares mensuales =
COUNTROWS(Cambios)

// Cambio precio mensual
Cambio precio mensual =
AVERAGE(Cambios[Cambio_precio])

// Cambio crudo mensual
Cambio crudo mensual =
AVERAGE(Cambios[Cambio_crudo])

// Pearson sobre cambios mensuales pareados. Se oculta con n<12; no demuestra causalidad ni independencia temporal.
Correlacion cambios =
VAR T = Cambios
VAR N = COUNTROWS(T)
VAR SX = SUMX(T,Cambios[Cambio_precio])
VAR SY = SUMX(T,Cambios[Cambio_crudo])
VAR SXY = SUMX(T,Cambios[Cambio_precio]*Cambios[Cambio_crudo])
VAR SXX = SUMX(T,Cambios[Cambio_precio]^2)
VAR SYY = SUMX(T,Cambios[Cambio_crudo]^2)
VAR D = MAX(0,(N*SXX-SX^2)*(N*SYY-SY^2))
RETURN IF(N>=12,DIVIDE(N*SXY-SX*SY,SQRT(D)))

// Aviso pares
Aviso pares =
IF([N pares mensuales]<12,"PRECAUCIÓN: menos de 12 pares; r se oculta.","Asociación descriptiva; no identifica causalidad ni rentabilidad.")

// Meses cohorte
Meses cohorte =
COUNTROWS(Cohorte)

// Celdas cobertura
Celdas cobertura =
COUNTROWS(Cobertura)

// Meses esperados por activo
Meses esperados por activo =
CALCULATE(COUNTROWS(Cobertura), KEEPFILTERS(Cobertura[Estado] <> "fuera de cobertura observada"))

// Meses con crudo válido / meses entre primer y último registro del activo. No penaliza meses fuera de su cobertura observada.
Cobertura crudo pct =
DIVIDE(SUM(Cobertura[Crudo_valido]), [Meses esperados por activo])

// Color cobertura
Color cobertura =
VAR P = [Cobertura crudo pct] RETURN IF(ISBLANK(P),"#E8EDF5",IF(P>=0.99,"#A4EAD6",IF(P>=0.90,"#FFE0A0","#FFADB9")))

// Activos sin catalogo
Activos sin catalogo =
SUMX(VALUES(Activos[Activo]), IF(CALCULATE([N registros])>0,CALCULATE(MAX(Activos[Catalogo_faltante])),0))

// Huecos visibles
Huecos visibles =
IF([Huecos internos]>0,[Huecos internos])

// Conflictos crudo visibles
Conflictos crudo visibles =
IF([Conflictos crudo]>0,[Conflictos crudo])

// Conflictos gas visibles
Conflictos gas visibles =
IF([Conflictos gas]>0,[Conflictos gas])
